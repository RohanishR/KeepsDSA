# Pattern: Hash Map (Two Pass or One Pass)
# Time Complexity: O(N)
# Space Complexity: O(N)

class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        prevMap = {} # val -> index

        for i, n in enumerate(nums):
            diff = target - n
            if diff in prevMap:
                return [prevMap[diff], i]
            prevMap[n] = i
        return []
